103: function ask(prompt: string): Promise<string> {
104:   // Queued lines and the closed-EOF case must be checked, and returned,
105:   // BEFORE touching `rl` at all: piped stdin can hit EOF (closing `rl`)
106:   // well before we get back around to the next ask() call — e.g. while a
107:   // real multi-second LLM turn is still streaming — and rl.prompt() throws
108:   // ERR_USE_AFTER_CLOSE if the interface is already closed. Only call it
109:   // once we know we're actually about to wait live, which is the one case
110:   // where rl is guaranteed still open (closed can't flip mid-synchronous-check
111:   // in a single-threaded event loop).
112:   if (lineQueue.length > 0) {
113:     const line = lineQueue.shift()!
114:     finishLine()
115:     return Promise.resolve(line)
116:   }
117:   if (closed) {
118:     finishLine()
119:     return Promise.resolve("/exit")
120:   }
121:   rl.setPrompt(prompt)
122:   rl.prompt()
123:   return new Promise((resolve) => {
124:     waitingResolve = (line) => {
125:       finishLine()
126:       resolve(line)
127:     }
128:   })
129: }
